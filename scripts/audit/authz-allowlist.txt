# Authorization allowlist — routes that are INTENTIONALLY public or authn-only.
# Format: one "METHOD /path" per line. '#' starts a comment.
# Read by scripts/audit/authz-coverage.ts. Keep this list small and justified.
#
# NOTE: pre-existing authn-only routes that still NEED guard-level authorization
# live in the separate authz-baseline.txt (the triage worklist), not here. Only
# add a route here when it is *correct* for it to lack a guard-level decorator —
# because it is pre-login, it acts only on the caller, it enforces authorization
# IN THE SERVICE LAYER (verified by inspection), or it is an authenticated
# tenant-scoped reference read.
#
# Each route below was read end-to-end; the justification states WHERE the
# authorization decision is made when it is not a guard.

# ── Pre-login / unauthenticated by necessity ──────────────────────────────
POST /auth/login              # credentials exchange; must be reachable before auth
GET /auth/health              # liveness probe
GET /tenants/lookup           # subdomain -> tenant resolution, used before login

# ── Authenticated-only by design (acts only on the caller, no resource to authz) ─
POST /auth/logout             # ends the caller's own session
POST /auth/refresh            # refreshes the caller's own token
GET /auth/me                  # returns the caller's own identity

# ── The Policy Decision Point itself — returns allow/deny, authn-only is correct ─
POST /users/:id/roles/permissions/check
GET /users/:id/roles/permissions/catalog   # static PERMISSION_REGISTRY catalog; no tenant data

# ── In-handler enforced: role-assignment delegation (verified in roles.service) ─
# Authorization is TenantAdmin OR school-Principal, with seniority-based
# escalation prevention (cannot assign/change a role at or above your own
# seniority; only TenantAdmin assigns Principal). This is RICHER than any single
# guard could express — a blanket @RequireGlobalRole('TenantAdmin') would break
# the legitimate Principal-assigns-Teacher delegation flow. Keep service-layer.
POST /users/:id/roles
PATCH /users/:id/roles/:schoolId
POST /users/:id/roles/:schoolId/change
DELETE /users/:id/roles/:schoolId
GET /users/:id/roles
GET /users/:id/roles/:schoolId

# ── In-handler enforced: resource-lookup-then-authz (verified in service code) ─
# The URL doesn't carry a schoolId — the FinanceJob row is the source of
# truth for scope. The controller: (1) loads the row via
# financeJobsService.get(jobId, context) which applies the same tenant
# isolation as every other GetItem in this service, and returns null on
# missing OR cross-school (preserving the 404-not-403 non-enumerability
# contract); (2) then, for non-TenantAdmin callers, evaluates billing:view
# against the identity RBAC matrix using the LOADED job's schoolId, same
# gate PermissionGuard would have run if the schoolId had been in the URL.
# Fails closed on identity transport error (same 404 shape as a deny).
# See bulk-ops.controller.ts:76-125 for the full end-to-end flow.
# A guard-level @RequirePermission decorator can't express "authz depends on
# the resource I haven't loaded yet" without either duplicating the DDB
# GetItem in the guard OR restructuring the URL to include schoolId (which
# would defeat the deep-link ergonomic from MVP.5's 409 body's runningJobId).
GET /finance/jobs/:jobId
